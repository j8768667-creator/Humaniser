  function ensureContext() {
    if (!ctx) {
      const AC = window.AudioContext || window.webkitAudioContext;
      ctx = new AC();
    }
    if (ctx.state === 'suspended') ctx.resume();
    return ctx;
  }

  // Unlock all file types toggle
  const defaultAccept = $('file').getAttribute('accept');
  $('unlockAllFiles').addEventListener('change', e => {
    if (e.target.checked) {
      $('file').removeAttribute('accept');
      setStatus('All file types unlocked');
    } else {
      $('file').setAttribute('accept', defaultAccept);
      setStatus('Audio filter active');
    }
  });

  // Drag & drop
  const dropZone = $('dropZone');
  dropZone.addEventListener('dragover', e => { e.preventDefault(); });
  dropZone.addEventListener('drop', e => {
    e.preventDefault();
    if (e.dataTransfer.files && e.dataTransfer.files[0]) handleFile(e.dataTransfer.files[0]);
  });

  $('file').addEventListener('change', e => {
    const f = e.target.files && e.target.files[0];
    if (f) handleFile(f);
  });

  async function handleFile(f) {
    try {
      stopPlayback();
      pauseOffset = 0;
      setStatus('Decoding audio…');
      $('fileInfo').textContent = `${f.name} · ${(f.size / 1048576).toFixed(2)} MB`;

      const ac = ensureContext();
      const bytes = await f.arrayBuffer();
      sourceBuffer = await ac.decodeAudioData(bytes.slice(0));
      processedBuffer = null;
      lastExportBlob = null;

      const baseName = f.name.replace(/\.[^/.]+$/, '');
      lastExportFilename = `${baseName}-deai-v2.wav`;

      enableTransport();
      drawWave(sourceBuffer, 0);
      updateTimeReadout(0, sourceBuffer.duration);
      $('renderStatus').textContent = 'Ready to render & export';
      setStatus(`Loaded · ${sourceBuffer.duration.toFixed(1)}s · ${sourceBuffer.sampleRate} Hz`);
    } catch (err) {
      console.error(err);
      setStatus('Decode error');
      $('fileInfo').textContent = 'Could not decode file. Try WAV, MP3, or M4A.';
    }
  }

  // Synthetic AI Test Voice Generator (deliberately robotic, static pitch + 7kHz vocoder buzz)
  $('loadDemo').addEventListener('click', () => {
    stopPlayback();
    pauseOffset = 0;
    const ac = ensureContext();
    const sr = ac.sampleRate;
    const dur = 5.0;
    const len = Math.floor(sr * dur);
    const buf = ac.createBuffer(1, len, sr);
    const d = buf.getChannelData(0);

    // Melodic phrase with dead-flat robotic pitch and harsh high-frequency vocoder buzz
    const melody = [146.83, 164.81, 174.61, 196.00, 164.81, 146.83];
    for (let i = 0; i < len; i++) {
      const t = i / sr;
      const step = Math.min(melody.length - 1, Math.floor(t / (dur / melody.length)));
      const f0 = melody[step];
      let sample = 0;

      // Rigid harmonic stack with metallic upper partials
      for (let h = 1; h <= 32; h++) {
        const freq = f0 * h;
        const f1 = Math.exp(-Math.pow((freq - 500) / 150, 2));
        const f2 = 0.7 * Math.exp(-Math.pow((freq - 1550) / 220, 2));
        const f3 = 0.5 * Math.exp(-Math.pow((freq - 2900) / 300, 2));
        // Artificial neural vocoder buzz peak around 6.8 kHz
        const aiBuzz = 0.45 * Math.exp(-Math.pow((freq - 6800) / 900, 2));
        const amp = (f1 + f2 + f3 + aiBuzz + 0.05) / Math.pow(h, 0.68);
        sample += Math.sin(2 * Math.PI * freq * t) * amp;
      }

      const env = Math.min(1, t / 0.05) * Math.min(1, (dur - t) / 0.15);
      d[i] = Math.tanh(sample * 0.42 * env);
    }

    sourceBuffer = buf;
    processedBuffer = null;
    lastExportBlob = null;
    lastExportFilename = 'synthetic-demo-deai.wav';
    $('fileInfo').textContent = 'Synthetic AI Vocal Test (5.0s · Robotic Pitch + 6.8kHz Vocoder Buzz)';
    enableTransport();
    drawWave(sourceBuffer, 0);
    updateTimeReadout(0, sourceBuffer.duration);
    $('renderStatus').textContent = 'Ready to render & export';
    setStatus('Compare ▶ Original vs ▶ Humanised!');
  });

  function drawWave(buf, playheadRatio = 0) {
    const c = $('wave');
    const g = c.getContext('2d');
    const w = c.width, h = c.height;
    g.clearRect(0, 0, w, h);
    g.fillStyle = '#0d1118';
    g.fillRect(0, 0, w, h);

    const mid = h / 2;
    g.strokeStyle = '#232e40';
    g.lineWidth = 1;
    g.beginPath();
    g.moveTo(0, mid);
    g.lineTo(w, mid);
    g.stroke();

    if (!buf) return;

    const data = buf.getChannelData(0);
    const step = Math.max(1, Math.floor(data.length / w));
    const headX = Math.floor(playheadRatio * w);

    for (let x = 0; x < w; x++) {
      let mn = 1, mx = -1;
      const start = x * step;
      for (let j = 0; j < step; j++) {
        const v = data[start + j] || 0;
        if (v < mn) mn = v;
        if (v > mx) mx = v;
      }
      g.strokeStyle = x <= headX ? '#7ee8be' : '#3a6657';
      g.lineWidth = 1.5;
      g.beginPath();
      g.moveTo(x, mid + mn * mid * 0.88);
      g.lineTo(x, mid + mx * mid * 0.88);
      g.stroke();
    }

    if (playheadRatio > 0 && playheadRatio < 1) {
      g.strokeStyle = '#f5c478';
      g.lineWidth = 2.2;
      g.beginPath();
      g.moveTo(headX, 0);
      g.lineTo(headX, h);
      g.stroke();
    }
  }

  // Waveform Scrubbing

  function scrubToEvent(e) {
    if (!sourceBuffer) return;

    const clientX = e.touches ? e.touches[0].clientX : e.clientX;
    const ratio = Math.max(0, Math.min(1, (clientX - rect.left) / rect.width));
    pauseOffset = ratio * sourceBuffer.duration;

    if (isPlaying) startLivePlayback(playMode, pauseOffset);
  }

  // Asymmetric Class-A Tube Saturation Curve (adds rich 2nd harmonic + vocal fry density)
  function makeAsymmetricTubeCurve(driveAmt) {
    const n = 8192;
    const curve = new Float32Array(n);
    const k = 1 + driveAmt * 12;
    const asym = 0.18 * driveAmt; // Even-harmonic asymmetry
    const norm = Math.tanh(k);
    for (let i = 0; i < n; i++) {
      const x = (i * 2) / n - 1;
      const biased = x + asym * (x * x - 0.35);
      curve[i] = Math.tanh(biased * k) / norm;
    }
    return curve;
  }

  // Early-Reflection Studio Booth Impulse
  function makeStudioBoothImpulse(ac, duration, decay) {
    const sr = ac.sampleRate;
    const len = Math.max(1, Math.floor(sr * duration));
    const imp = ac.createBuffer(2, len, sr);
    let seed = 424242;
    const rand = () => {
      seed = (seed * 16807) % 2147483647;
      return (seed - 1) / 1073741823.5 - 1;
    };
    const earlyTaps = [0.007, 0.013, 0.019, 0.027, 0.035];
    for (let ch = 0; ch < 2; ch++) {
      const d = imp.getChannelData(ch);
      // Discrete wood booth early reflections
      earlyTaps.forEach((tapSec, idx) => {
        const samp = Math.floor((tapSec + ch * 0.0015) * sr);
        if (samp < len) d[samp] = (idx % 2 === 0 ? 0.5 : -0.45) * Math.pow(0.78, idx);
      });
      const tailStart = Math.floor(0.015 * sr);
      for (let i = tailStart; i < len; i++) {
        const t = (i - tailStart) / (len - tailStart);
        d[i] += rand() * Math.pow(1 - t, decay) * 0.32;
      }
    }
    return imp;
  }

  /**
   * V2 DEEP DE-AI VOCAL RESYNTHESIS GRAPH
   * 100% Serial Pitch Drift + Dynamic Formant Filter Bank + Envelope-Gated Breath Turbulence + Codec De-Buzzer
   */
  function buildGraph(ac, inputNode, initialSettings, bypass = false) {
    const dryOut = ac.createGain();
    const wetOut = ac.createGain();
    const masterOut = ac.createGain();

    // Bypass path
    inputNode.connect(dryOut);
    dryOut.connect(masterOut);

    // 1. High-Pass Rumble Filter
    const hp = ac.createBiquadFilter();
    hp.type = 'highpass';
    hp.frequency.value = 78;
    hp.Q.value = 0.7;

    // 2. 100% SERIAL PITCH DRIFT & HUMAN VIBRATO STAGE
    // Because the entire signal goes through this delay line serially (no parallel dry voice),
    // modulating delayTime physically bends the pitch of the voice without flanging/chorusing!
    const serialPitchDelay = ac.createDelay(0.1);
    serialPitchDelay.delayTime.value = 0.024;

    const driftLFO1 = ac.createOscillator(); // Slow phrase pitch drift (0.37 Hz)
    const driftLFO2 = ac.createOscillator(); // Medium syllable drift (1.15 Hz)
    const vibratoLFO = ac.createOscillator(); // Natural human larynx vibrato (5.4 Hz)
    driftLFO1.type = 'sine';
    driftLFO2.type = 'triangle';
    vibratoLFO.type = 'sine';

    const driftGain1 = ac.createGain();
    const driftGain2 = ac.createGain();
    const vibratoGain = ac.createGain();

    driftLFO1.connect(driftGain1);
    driftLFO2.connect(driftGain2);
    vibratoLFO.connect(vibratoGain);

    driftGain1.connect(serialPitchDelay.delayTime);
    driftGain2.connect(serialPitchDelay.delayTime);
    vibratoGain.connect(serialPitchDelay.delayTime);

    driftLFO1.start(0);
    driftLFO2.start(0);
    vibratoLFO.start(0);

    // 3. DYNAMIC FORMANT TRACT SHIFTER (F1 & F2 Moving Cavity Filters)
    // Modulates throat (550 Hz) and mouth (1650 Hz) resonances in opposite directions
    const formant1 = ac.createBiquadFilter();
    formant1.type = 'peaking';
    formant1.frequency.value = 540;
    formant1.Q.value = 2.2;

    const formant2 = ac.createBiquadFilter();
    formant2.type = 'peaking';
    formant2.frequency.value = 1680;
    formant2.Q.value = 2.5;

    const formantLFO = ac.createOscillator();
    formantLFO.type = 'sine';
    formantLFO.frequency.value = 0.78;
    const f1ModGain = ac.createGain();
    const f2ModGain = ac.createGain();
    formantLFO.connect(f1ModGain);
    formantLFO.connect(f2ModGain);
    f1ModGain.connect(formant1.frequency);
    f2ModGain.connect(formant2.frequency);
    formantLFO.start(0);

    // 4. CHEST WEIGHT & PLASTIC BOXINESS CUT
    const chestShelf = ac.createBiquadFilter();
    chestShelf.type = 'peaking';
    chestShelf.frequency.value = 165;
    chestShelf.Q.value = 0.85;

    const boxCut = ac.createBiquadFilter();
    boxCut.type = 'peaking';
    boxCut.frequency.value = 390;
    boxCut.Q.value = 1.4;

    // 5. ASYMMETRIC TUBE SATURATION & HARMONIC EXCITER
    const tubeShaper = ac.createWaveShaper();
    tubeShaper.oversample = '4x';

    // 6. DYNAMIC BREATH / RASP MODULATION (Signal-Multiplied Air Turbulence)
    // By modulating a high-passed vocal path at audio-rate noise frequencies,
    // breath air is generated ONLY when the singer is actually making sound (zero noise in silence).
    const breathBP = ac.createBiquadFilter();
    breathBP.type = 'bandpass';
    breathBP.frequency.value = 2800;
    breathBP.Q.value = 0.9;

    const breathShaper = ac.createWaveShaper();
    // High-order gritty rasp transfer curve
    const raspCurve = new Float32Array(4096);
    for (let i = 0; i < 4096; i++) {
      const x = (i * 2) / 4096 - 1;
      raspCurve[i] = Math.sin(x * Math.PI * 3.5) * (1 - Math.abs(x) * 0.4);
    }
    breathShaper.curve = raspCurve;
    const breathGain = ac.createGain();

    // 7. AI CODEC DE-BUZZER (Dual Notch + Warm Tape Roll-Off)
    // Erases 6.4 kHz and 8.2 kHz metallic HiFi-GAN / neural vocoder phase ringing
    const debuzzNotch1 = ac.createBiquadFilter();
    debuzzNotch1.type = 'peaking';
    debuzzNotch1.frequency.value = 6350;
    debuzzNotch1.Q.value = 2.1;

    const debuzzNotch2 = ac.createBiquadFilter();
    debuzzNotch2.type = 'peaking';
    debuzzNotch2.frequency.value = 8400;
    debuzzNotch2.Q.value = 2.4;

    const tapeLowpass = ac.createBiquadFilter();
    tapeLowpass.type = 'lowpass';
    tapeLowpass.frequency.value = 13500;
    tapeLowpass.Q.value = 0.6;

    // 8. CONDENSER MIC PROXIMITY & PRESENCE
    const micPresence = ac.createBiquadFilter();
    micPresence.type = 'peaking';
    micPresence.frequency.value = 2750;
    micPresence.Q.value = 0.9;

    // 9. OPTICAL VOCAL COMPRESSOR
    const comp = ac.createDynamicsCompressor();
    comp.knee.value = 12;
    comp.attack.value = 0.008;
    comp.release.value = 0.15;
    const makeupGain = ac.createGain();

    // 10. ACOUSTIC MICRO-DOUBLER & STUDIO BOOTH
    const doublerDelay = ac.createDelay(0.06);
    doublerDelay.delayTime.value = 0.017;
    const doublerGain = ac.createGain();

    const boothConvolver = ac.createConvolver();
    boothConvolver.buffer = makeStudioBoothImpulse(ac, 0.52, 3.4);
    const boothLP = ac.createBiquadFilter();
    boothLP.type = 'lowpass';
    boothLP.frequency.value = 3600;
    const boothGain = ac.createGain();

    // Brickwall output safety limiter
    const limiter = ac.createDynamicsCompressor();
    limiter.threshold.value = -1.0;
    limiter.knee.value = 1.5;
    limiter.ratio.value = 20.0;
    limiter.attack.value = 0.001;
    limiter.release.value = 0.06;

    // Connect Serial Chain
    inputNode.connect(hp);
    hp.connect(serialPitchDelay);
    serialPitchDelay.connect(formant1);
    formant1.connect(formant2);
    formant2.connect(chestShelf);
    chestShelf.connect(boxCut);
    boxCut.connect(tubeShaper);

    // Dynamic Breath / Rasp parallel injector (strictly envelope-locked)
    serialPitchDelay.connect(breathBP);
    breathBP.connect(breathShaper);
    breathShaper.connect(breathGain);

    const postDriveSum = ac.createGain();
    tubeShaper.connect(postDriveSum);
    breathGain.connect(postDriveSum);

    postDriveSum.connect(micPresence);
    micPresence.connect(debuzzNotch1);
    debuzzNotch1.connect(debuzzNotch2);
    debuzzNotch2.connect(tapeLowpass);
    tapeLowpass.connect(comp);
    comp.connect(makeupGain);

    // Spatial / Booth connections
    makeupGain.connect(wetOut);

    makeupGain.connect(doublerDelay);
    doublerDelay.connect(doublerGain);
    doublerGain.connect(wetOut);

    makeupGain.connect(boothConvolver);
    boothConvolver.connect(boothLP);
    boothLP.connect(boothGain);
    boothGain.connect(wetOut);

    wetOut.connect(limiter);
    limiter.connect(masterOut);

    function updateParams(s, isBypassed = false) {
      const now = ac.currentTime;
      if (isBypassed) {
        dryOut.gain.setTargetAtTime(1, now, 0.015);
        wetOut.gain.setTargetAtTime(0, now, 0.015);
        return;
      }

      dryOut.gain.setTargetAtTime(0, now, 0.015);
      wetOut.gain.setTargetAtTime(1, now, 0.015);

      const master = 0.35 + (s.amount / 100) * 0.95;

      // 1. Serial Pitch & Vibrato Drift
      const p = (s.pitch / 100) * master;
      driftLFO1.frequency.setTargetAtTime(0.34 + (s.amount / 100) * 0.15, now, 0.02);
      driftLFO2.frequency.setTargetAtTime(1.18 + (s.pitch / 100) * 0.35, now, 0.02);
      vibratoLFO.frequency.setTargetAtTime(5.35, now, 0.02);

      // Delay time modulation depth -> directly creates pitch deviation (Δf/f = -dD/dt)
      driftGain1.gain.setTargetAtTime(p * 0.0032, now, 0.02);
      driftGain2.gain.setTargetAtTime(p * 0.0011, now, 0.02);
      vibratoGain.gain.setTargetAtTime(p * 0.00042, now, 0.02);

      // 2. Dynamic Throat Formant Shifting
      const fm = (s.formant / 100) * master;
      formant1.gain.setTargetAtTime(fm * 5.2, now, 0.02);
      formant2.gain.setTargetAtTime(-fm * 4.4, now, 0.02);
      f1ModGain.gain.setTargetAtTime(fm * 145, now, 0.02);   // Swings F1 +/- 145 Hz
      f2ModGain.gain.setTargetAtTime(-fm * 290, now, 0.02);  // Counter-swings F2 +/- 290 Hz

      // 3. Chest Weight & Boxiness Cut
      const w = (s.warmth / 100) * master;
      chestShelf.gain.setTargetAtTime(w * 6.8, now, 0.02);
      boxCut.gain.setTargetAtTime(-w * 5.2, now, 0.02);

      // 4. Asymmetric Tube Drive
      const tx = (s.texture / 100) * master;
      tubeShaper.curve = makeAsymmetricTubeCurve(tx * 0.65);

      // 5. Envelope-Locked Breath & Vocal Rasp
      const br = (s.breath / 100) * master;
      breathGain.gain.setTargetAtTime(br * 0.16, now, 0.02);

      // 6. AI Codec De-Buzzer (6.35kHz + 8.4kHz suppression + analog tape roll-off)
      const db = (s.debuzz / 100) * master;
      debuzzNotch1.gain.setTargetAtTime(-db * 11.5, now, 0.02);
      debuzzNotch2.gain.setTargetAtTime(-db * 10.0, now, 0.02);
      const cutoff = Math.max(6200, 16500 - db * 8200);
      tapeLowpass.frequency.setTargetAtTime(cutoff, now, 0.02);

      // 7. Condenser Mic Presence (2.75 kHz)
      const pr = (s.presence / 100) * master;
      micPresence.gain.setTargetAtTime(pr * 5.5 - 1.0, now, 0.02);

      // 8. Optical Compression
      const dyn = (s.dynamics / 100) * master;
      comp.threshold.setTargetAtTime(-10 - dyn * 20, now, 0.02);
      comp.ratio.setTargetAtTime(1.5 + dyn * 4.5, now, 0.02);
      makeupGain.gain.setTargetAtTime(1.0 + dyn * 0.55, now, 0.02);

      // 9. Acoustic Doubler & Studio Booth
      const sp = (s.space / 100) * master;
      doublerGain.gain.setTargetAtTime(sp * 0.26, now, 0.02);
      boothGain.gain.setTargetAtTime(sp * 0.45, now, 0.02);
    }

    updateParams(initialSettings, bypass);

    return {
      output: masterOut,
      updateParams,
      dispose() {
        try {
          driftLFO1.stop();
          driftLFO2.stop();
          vibratoLFO.stop();
          formantLFO.stop();
        } catch (e) {}
        try { masterOut.disconnect(); } catch (e) {}
      }
    };
  }

  // Real-Time Playback Engine
  function stopPlayback(resetOffset = true) {
    if (rafId) {
      cancelAnimationFrame(rafId);
      rafId = null;
    }
    if (activeSource) {
      activeSource.onended = null;
      try { activeSource.stop(); } catch (e) {}
      try { activeSource.disconnect(); } catch (e) {}
      activeSource = null;
    }
    if (activeGraph) {
      activeGraph.dispose();
      activeGraph = null;
    }
    isPlaying = false;
    $('meterFill').style.width = '0%';
    $('playOriginal').classList.remove('active-mode');
    $('playHuman').classList.remove('active-mode');
    if (resetOffset && sourceBuffer) {
      pauseOffset = 0;
      drawWave(sourceBuffer, 0);
      updateTimeReadout(0, sourceBuffer.duration);
    }
  }

  function startLivePlayback(mode, offset = pauseOffset) {
    if (!sourceBuffer) return;
    const ac = ensureContext();

    // Instant A/B switching while playing
    if (isPlaying && activeGraph && offset === pauseOffset) {
      playMode = mode;
      activeGraph.updateParams(getSettings(), playMode === 'original');
      $('playOriginal').classList.toggle('active-mode', playMode === 'original');
      $('playHuman').classList.toggle('active-mode', playMode === 'human');
      setStatus(playMode === 'original' ? 'Playing Original (Raw AI)' : 'Playing Humanised (Live De-AI)');
      return;
    }

    stopPlayback(false);
    playMode = mode;

    const src = ac.createBufferSource();
    src.buffer = sourceBuffer;
    src.loop = isLooping;

    const graph = buildGraph(ac, src, getSettings(), mode === 'original');
    analyser = ac.createAnalyser();
    analyser.fftSize = 256;

    graph.output.connect(analyser);
    analyser.connect(ac.destination);

    activeSource = src;
    activeGraph = graph;
    isPlaying = true;

    const safeOffset = Math.max(0, Math.min(offset, sourceBuffer.duration - 0.05));
    startTime = ac.currentTime - safeOffset;

    $('playOriginal').classList.toggle('active-mode', mode === 'original');
    $('playHuman').classList.toggle('active-mode', mode === 'human');

    src.onended = () => {
      if (activeSource === src && !isLooping) {
        stopPlayback(true);
        setStatus('Playback finished');
      }
    };

    src.start(0, safeOffset);
    setStatus(mode === 'original' ? 'Playing Original (Raw AI)' : 'Playing Humanised (Live De-AI)');
    tickVisualizer();
  }

  function tickVisualizer() {
    if (!isPlaying || !sourceBuffer || !ctx) return;
    let elapsed = ctx.currentTime - startTime;
    if (isLooping) {
      elapsed = elapsed % sourceBuffer.duration;
    } else if (elapsed >= sourceBuffer.duration) {
      elapsed = sourceBuffer.duration;
    }
    pauseOffset = elapsed;
    drawWave(sourceBuffer, elapsed / sourceBuffer.duration);
    updateTimeReadout(elapsed, sourceBuffer.duration);

    if (analyser) {
      const td = new Uint8Array(analyser.fftSize);
      analyser.getByteTimeDomainData(td);
      let peak = 0;
      for (let i = 0; i < td.length; i++) {
        const v = Math.abs((td[i] - 128) / 128);
        if (v > peak) peak = v;
      }
      $('meterFill').style.width = Math.min(100, peak * 115).toFixed(1) + '%';
    }
    rafId = requestAnimationFrame(tickVisualizer);
  }

  function normalizeAudioBuffer(buf, targetPeak = 0.89125) {
    let maxPeak = 0;
    for (let c = 0; c < buf.numberOfChannels; c++) {
      const d = buf.getChannelData(c);
      for (let i = 0; i < d.length; i++) {
        const a = Math.abs(d[i]);
        if (a > maxPeak) maxPeak = a;
      }
    }
    if (maxPeak > 0.0001) {
      const gain = targetPeak / maxPeak;
      for (let c = 0; c < buf.numberOfChannels; c++) {
        const d = buf.getChannelData(c);
        for (let i = 0; i < d.length; i++) d[i] *= gain;
      }
    }
  }

  function encodeWAV(buf, bitDepth = 24) {
    const channels = buf.numberOfChannels;
    const rate = buf.sampleRate;
    const frames = buf.length;
    const bytesPerSample = bitDepth === 24 ? 3 : 2;
    const blockAlign = channels * bytesPerSample;
    const dataSize = frames * blockAlign;
    const ab = new ArrayBuffer(44 + dataSize);
    const v = new DataView(ab);

    function str(offset, s) {
      for (let i = 0; i < s.length; i++) v.setUint8(offset + i, s.charCodeAt(i));
    }

    str(0, 'RIFF');
    v.setUint32(4, 36 + dataSize, true);
    str(8, 'WAVE');
    str(12, 'fmt ');
    v.setUint32(16, 16, true);
    v.setUint16(20, 1, true);
    v.setUint16(22, channels, true);
    v.setUint32(24, rate, true);
    v.setUint32(28, rate * blockAlign, true);
    v.setUint16(32, blockAlign, true);
    v.setUint16(34, bitDepth, true);
    str(36, 'data');
    v.setUint32(40, dataSize, true);

    const chData = Array.from({ length: channels }, (_, c) => buf.getChannelData(c));
    let off = 44;

    if (bitDepth === 24) {
      for (let i = 0; i < frames; i++) {
        for (let c = 0; c < channels; c++) {
          const x = Math.max(-1, Math.min(1, chData[c][i]));
          const sample = Math.round(x < 0 ? x * 8388608 : x * 8388607);
          v.setUint8(off, sample & 0xff);
          v.setUint8(off + 1, (sample >> 8) & 0xff);
          v.setUint8(off + 2, (sample >> 16) & 0xff);
          off += 3;
        }
      }
    } else {
      for (let i = 0; i < frames; i++) {
        for (let c = 0; c < channels; c++) {
          const x = Math.max(-1, Math.min(1, chData[c][i]));
          v.setInt16(off, x < 0 ? x * 32768 : x * 32767, true);
          off += 2;
        }
      }
    }
    return new Blob([ab], { type: 'audio/wav' });
  }

  function triggerDownload(blob, filename) {
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.style.display = 'none';
    a.href = url;
    a.download = filename;
    document.body.appendChild(a);
    a.click();
    setTimeout(() => {
      a.remove();
      URL.revokeObjectURL(url);
    }, 60000);
  }

  $('render').addEventListener('click', async () => {
    if (!sourceBuffer) return;
    $('render').disabled = true;
    $('renderStatus').textContent = 'Rendering De-AI master…';
    setStatus('Processing offline render…');

    try {
      const s = getSettings();
      const tailSeconds = s.space > 2 ? 0.55 : 0.12;
      const totalFrames = sourceBuffer.length + Math.ceil(sourceBuffer.sampleRate * tailSeconds);
      const Offline = window.OfflineAudioContext || window.webkitOfflineAudioContext;
      const off = new Offline(sourceBuffer.numberOfChannels, totalFrames, sourceBuffer.sampleRate);

      const src = off.createBufferSource();
      src.buffer = sourceBuffer;

      const graph = buildGraph(off, src, s, false);
      graph.output.connect(off.destination);
      src.start(0);

      processedBuffer = await off.startRendering();

      if ($('normalizeChk').checked) {
        normalizeAudioBuffer(processedBuffer, 0.89125);
      }

      const bits = parseInt($('bitDepth').value, 10) || 24;
      lastExportBlob = encodeWAV(processedBuffer, bits);

      $('shareBtn').disabled = false;
      $('renderStatus').textContent = `Exported ${bits}-bit WAV (${(lastExportBlob.size / 1048576).toFixed(1)} MB)`;
      setStatus('De-AI WAV exported');

      triggerDownload(lastExportBlob, lastExportFilename);
    } catch (err) {
      console.error(err);
      $('renderStatus').textContent = 'Render failed';
      setStatus('Error — try a shorter clip');
    } finally {
      $('render').disabled = false;
    }
  });

  $('shareBtn').addEventListener('click', async () => {
    if (!lastExportBlob) return;
    const file = new File([lastExportBlob], lastExportFilename, { type: 'audio/wav' });
    if (navigator.canShare && navigator.canShare({ files: [file] })) {
      try {
        await navigator.share({ files: [file], title: lastExportFilename });
        setStatus('Shared via iPad Share Sheet');
        return;
      } catch (err) {
        if (err.name === 'AbortError') return;
      }
    }
    triggerDownload(lastExportBlob, lastExportFilename);
    setStatus('Downloaded WAV');
  });

  drawWave(null, 0);
  setVals(presets.killai);
})();
